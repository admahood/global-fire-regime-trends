// create modal flammable lands
// flammable lands
var flammable = lc.map(function(image){
  return image.remap(
    [11, 13, 15, 16, 17], 
    [0, 0, 0, 0, 0], 1); //remap, all others 1
})
  .reduce(ee.Reducer.mode())
  .rename('flammable');
// visualize
Map.addLayer(
  flammable, {min:0,max:1}, 
  "Flammable Lands (2003-2020)", false);


/**
* 1a. Percent area of flammable landscapes
* 1b. gHM mean score by country
* 1c. gHM mean score within flammable landscapes
*/


// Visualize the gHM
ghm = ghm.select('gHM').first().clipToCollection(countries);
Map.addLayer(ghm, ghmVis, 'Human Modification', false);


// Now, multiply flammable lands against the GHM
var ghm_flammable = 
  flammable.multiply(ghm).rename('gHM_flammable');
Map.addLayer(
  ghm_flammable, {}, "Human Modification w/in Flammable Lands", false);


// Stack together
var stack = ghm.addBands([flammable, ghm_flammable]);
// print(stack.bandNames());


// Loop through countries, export
var reduc = countries.map(function(ftr){
  
  // Calculate the area of burnable land (km2)
  var areaImage = flammable.multiply(ee.Image.pixelArea());
  
  // Sum reducer
  var area = areaImage.reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: ftr.geometry(),
    scale: 500,
    maxPixels: 1e13
  });
  
  // convert to km2
  var flammableAreaSqKm = 
    ee.Number(area.get('flammable'))
    .divide(1e6).round();
  
  // get the percent area for the country
  var percent = 
    flammableAreaSqKm.divide(
      ftr.geometry().area().divide(1000000)
    ).multiply(100);
  
  area = area.set('percent_flammable', percent);
  
  var meanReduc = stack
    .select(['gHM', 'gHM_flammable'])
    .reduceRegion({
      reducer: ee.Reducer.mean(),
      geometry: ftr.geometry(),
      scale: 1000,
      maxPixels: 1e13,
      tileScale: 16
    });
  
  var stdevReduc = stack
    .select(['gHM', 'gHM_flammable'])
    .reduceRegion({
      reducer: ee.Reducer.stdDev(),
      geometry: ftr.geometry(),
      scale: 1000,
      maxPixels: 1e13,
      tileScale: 16
    });
  
  var merge = 
    area.combine(meanReduc).combine(stdevReduc);
  
  // Submit the output dictionary to a feature
  var dict_to_ftr = ee.Feature(null, merge); 
  var output_ftr = ee.Feature(ftr.copyProperties(dict_to_ftr));
  
  // Return the final merged table
  return(output_ftr);
  
});

// select the bands of interest
reduc = 

  reduc
    .select(
      ['aoi', 'perc_flammable', 'gHM', 'gHM_flammable_mean']
    );

//Check it worked
Map.addLayer(
  reduc, {}, "gHM/Flammable Summary by Country", false);

// Export the table
Export.table.toDrive({
  collection: reduc,
  description: 'human_mod_inFlammable-static',
  folder: 'global-fire-report',
  fileFormat: 'GeoJSON'
});
