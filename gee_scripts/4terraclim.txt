/**
 * Summaries of TerraClim VPD by country.
 * 
 * We summarize Vapor Pressure Deficit (VPD) within 'flammable lands'
 *  at both the annual time step and for the peak burn season of each country
 *
 * Data Sources:
 * - TerraClimate (Abatzaglou et al. 2018)
 * - MCD12Q1 (MODIS Land Cover, Type 1)
 * 
 * Author:
 * Maxwell C. Cook, PhD Student
 * Department of Geography,
 * University of Colorado Boulder
 * maxwell.cook@colorado.edu
 *

Map.setOptions('hybrid');

// grab a water mask
var water = mcd12q1.select('LC_Type1').mode().neq(17);

// Create the flammable area mask
// Grab the MCD12Q1 Type 1 Land Cover
var flammable = mcd12q1
  .select('LC_Type1')
  // Filter to temporal range (2003-2020)
  .filter(ee.Filter.calendarRange(2003,2021,'year'))
  .map(function(image){
    return image.remap(
      [11, 13, 15, 16, 17], 
      [0, 0, 0, 0, 0], 1) //remap, all others 1
        .updateMask(water)
        .rename('flammable')
        .set(
          'system:time_start', 
          image.get('system:time_start'));
  }).reduce(ee.Reducer.mode())
    .rename('flammable');

// print("TerraClim ImageCollection: ", terraclim);

// Band names
var bands = terraclim.first().bandNames();
// print("TerraClim Bands: ", bands);

// Grab the res, crs
var proj = terraclim.first().select('vpd').projection();
var res = proj.nominalScale();

// mask the terraclim
// select just the VPD band
// mask to flammable lands
// be sure to reset the system:time_start
terraclim = terraclim.map(function(image){
  return image
            .select('vpd')
            .updateMask(flammable.eq(1))
            .set('system:time_start', image.get('system:time_start'));
});

/**
 * Work with monthly mean and anomoly + annual summaries
 * Including slicing to peak burn season months
 */

// subset to analysis time period
var imgc = terraclim
  .filter(ee.Filter.calendarRange(2003, 2021, 'year'));


// add the VPD anomoly as a band
// monthly pixel value - long-term monthly mean
imgc = imgc.map( function( image ) {
  
  // get the month of the image
  var m = ee.Number.parse(
    ee.Date(image.get('system:time_start')).format('MM')
  );
  
  // calculate the long-term monthly mean
  var ltmmean = terraclim
    .filter(ee.Filter.calendarRange(m,m,'month'))
    .mean();
  
  // calculate the anomoly
  var anom = image.subtract(ltmmean).toInt16().rename('vpd_anom');
  
  return image.addBands(anom)
              .set('system:time_start', image.get('system:time_start'));
              
});

// print(imgc.first().bandNames(), imgc);

// list of years
var years = ee.List.sequence(2003,2020,1);
var months = ee.List.sequence(1,12,1);

// Calculate annual statistics
var stat = ee.Reducer.mean();
// var stat = ee.Reducer.min();
// var stat = ee.Reducer.max();
// var stat = ee.Reducer.stdDev();

countries = countries.filter(ee.Filter.neq('start_mo', null));

// Loop through countries, export
var terrasum = countries.map(function( ftr ){
  
  // grab start and end of peak fire season
  var start_month = ftr.get('start_mo');
  var end_month = ftr.get('end_mo');
  
  // filter for the year / peak season
  var imgmo = ee.ImageCollection.fromImages(
    
    years.map(function( y ) {
    
      return imgc
        .filter(ee.Filter.calendarRange(y,y,'year'))
        .filter(ee.Filter.calendarRange(start_month,end_month,'month'))
        .mean() // take the mean peak season VPD and VPD anomoly
        // .min() // take the min
        // .max() // take the max
        // .reduce(ee.Reducer.stdDev()) // take the stDev
        .set('year', y)
        .set('start_month',start_month)
        .set('end_month',end_month)
        .toInt8();
      })
    );
  
  // stack the image collection as bands
  var stack = imgmo.toBands();
  
  // now calculate statistics
  // for all (min,max,mean,stdev) take the mean of the country
  var reduc = stack.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: ftr.geometry(),
    scale: res,
    maxPixels: 1e13,
    tileScale: 16,
    bestEffort: true
  });
  
  // Submit the output dictionary to a feature
  var dict_to_ftr = ee.Feature(null, reduc); 
  var output_ftr = ee.Feature(ftr.copyProperties(dict_to_ftr));
  
  // Return the final merged table
  return(output_ftr);
  
});
